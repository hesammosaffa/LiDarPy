cmake_minimum_required(VERSION 3.20)
project(rs_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)

set(RS_DRIVER_SRC "" CACHE PATH "folder that contains rs_driver/")
find_package(pybind11 CONFIG REQUIRED)
find_package(Threads REQUIRED)

pybind11_add_module(rs_bridge rs_bridge.cpp)
target_include_directories(rs_bridge PRIVATE ${RS_DRIVER_SRC})
target_compile_definitions(rs_bridge PRIVATE DISABLE_PCAP_PARSE)
if(MSVC)
  target_compile_options(rs_bridge PRIVATE /utf-8 /bigobj)
  target_compile_definitions(rs_bridge PRIVATE _WINSOCK_DEPRECATED_NO_WARNINGS NOMINMAX)
endif()
target_link_libraries(rs_bridge PRIVATE Threads::Threads)
if(WIN32)
  target_link_libraries(rs_bridge PRIVATE ws2_32)
endif()
